2. Model Design and Training

2.1 Define the CNN Model

We build a convolutional neural network with:

  • 3 convolutional blocks (Conv2d → BatchNorm → ReLU → MaxPool)
  • Dropout for regularization
  • 2 fully-connected layers
  • Softmax output for 10 classes
    • class ImageClassifier(nn.Module):
    """CNN for CIFAR-10 image classification."""

    def __init__(self):
        super(ImageClassifier, self).__init__()

        # Conv block 1: 3 -> 32 channels
        self.conv1 = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),       # 32x32 -> 16x16
        )

        # Conv block 2: 32 -> 64 channels
        self.conv2 = nn.Sequential(
            nn.Conv2d(32, 64, kernel_size=3, padding=1),
            nn.BatchNorm2d(64),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),       # 16x16 -> 8x8
        )

        # Conv block 3: 64 -> 128 channels
        self.conv3 = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.ReLU(),
            nn.MaxPool2d(2, 2),       # 8x8 -> 4x4
        )

        # Classifier
        self.classifier = nn.Sequential(
            nn.Dropout(0.4),
            nn.Linear(128 * 4 * 4, 256),
            nn.ReLU(),
            nn.Dropout(0.3),
            nn.Linear(256, 10),
        )

    def forward(self, x):
        x = self.conv1(x)
        x = self.conv2(x)
        x = self.conv3(x)
        x = x.view(x.size(0), -1)    # Flatten
        x = self.classifier(x)
        return x                      # Raw logits (CrossEntropyLoss applies softmax)

    def predict_proba(self, x):
        """Return softmax probabilities for each of the 10 classes."""
        logits = self.forward(x)
        return torch.softmax(logits, dim=1)

    2.3 Train the Model

    def train_model(model, trainloader, testloader, criterion, optimizer, scheduler, epochs=20):
    """Train the model and record loss/accuracy per epoch."""
    train_losses = []
    test_losses = []
    test_accuracies = []

    for epoch in range(epochs):
        # ── Training phase ──
        model.train()
        running_loss = 0.0
        for images, labels in trainloader:
            images, labels = images.to(device), labels.to(device)

            optimizer.zero_grad()
            outputs = model(images)
            loss = criterion(outputs, labels)
            loss.backward()
            optimizer.step()

            running_loss += loss.item()

        avg_train_loss = running_loss / len(trainloader)
        train_losses.append(avg_train_loss)

        # ── Evaluation phase ──
        model.eval()
        test_loss = 0.0
        correct = 0
        total = 0
        with torch.no_grad():
            for images, labels in testloader:
                images, labels = images.to(device), labels.to(device)
                outputs = model(images)
                loss = criterion(outputs, labels)
                test_loss += loss.item()
                _, predicted = torch.max(outputs, 1)
                total += labels.size(0)
                correct += (predicted == labels).sum().item()

        avg_test_loss = test_loss / len(testloader)
        accuracy = 100.0 * correct / total
        test_losses.append(avg_test_loss)
        test_accuracies.append(accuracy)

        # Step scheduler
        scheduler.step(avg_test_loss)
        current_lr = optimizer.param_groups[0]['lr']

        print(f"Epoch [{epoch+1:2d}/{epochs}] "
              f"Train Loss: {avg_train_loss:.4f} | "
              f"Test Loss: {avg_test_loss:.4f} | "
              f"Accuracy: {accuracy:.2f}% | "
              f"LR: {current_lr:.6f}")

    return train_losses, test_losses, test_accuracies

    3. Model Testing and Evaluation

    3.1 Evaluate on Test Set

    def evaluate_model(model, testloader):
    """Evaluate model on the test set and print per-class accuracy."""
    model.eval()
    correct = 0
    total = 0
    class_correct = [0] * 10
    class_total = [0] * 10

    with torch.no_grad():
        for images, labels in testloader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            _, predicted = torch.max(outputs, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()

            for i in range(labels.size(0)):
                label = labels[i].item()
                class_total[label] += 1
                if predicted[i] == label:
                    class_correct[label] += 1

    overall_acc = 100.0 * correct / total
    print(f"Overall Test Accuracy: {overall_acc:.2f}%\n")
    print(f"{'Class':<15} {'Accuracy':>10}")
    print("-" * 27)
    for i in range(10):
        acc = 100.0 * class_correct[i] / class_total[i] if class_total[i] > 0 else 0
        print(f"{classes[i]:<15} {acc:>9.2f}%")

    return overall_acc

overall_accuracy = evaluate_model(model, testloader)

    3.3 Analysis and Recommendation

    Our CNN model achieves 81.13% accuracy on the CIFAR-10 test set, which significantly exceeds the 45% minimum threshold required by the rubric. Compared to a hypothetical random-guess model that would achieve only 10% accuracy (1 out of 10 classes), our model performs roughly 8x better, demonstrating that the neural network has learned meaningful visual features from the training data. The per-class results show strong performance on vehicles (automobile: 93.7%, ship: 88.4%) while leaving room for improvement on visually similar animals (cat: 56.6%, bird: 63.5%).

    While this accuracy is strong for a relatively simple 3-block CNN trained from scratch, state-of-the-art models on CIFAR-10 (e.g., ResNets, EfficientNets with extensive augmentation) can exceed 95%. For a production object detection system, I would recommend building a custom solution using transfer learning with a pretrained backbone (such as ResNet-50), which would offer the best balance of performance, customizability, and cost—especially since the foundational PyTorch skills demonstrated here transfer directly to more advanced architectures.
